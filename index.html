Receive results automatically in a Google Sheet (optional)
  1. Create a new Google Sheet at sheets.google.com.
  2. Click Extensions → Apps Script. Delete anything in the editor, paste the code below and click Save.
  3. Click Deploy → New deployment, choose the gear icon → Web app.
  4. Set Execute as: Me and Who has access: Anyone, then Deploy. Approve the permissions (if Google warns you, click Advanced → Go to project).
  5. Copy the Web app URL, paste it in the box below and press Send test result. A "Maths Pairs Results" tab should appear in your sheet.
  6. Create your student link below. The URL is built into the link. If you later edit the script, use Deploy → Manage deployments → Edit → New version so the URL stays the same.
function doPost(e) {
  var d = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('Maths Pairs Results') || ss.insertSheet('Maths Pairs Results');
  if (sh.getLastRow() === 0) sh.appendRow(['Date','Name','Activity','Game mode','Pairs matched','Attempts','Time (s)','Skills']);
  sh.appendRow([new Date(), d.name, d.activity, d.mode, d.pairs, d.attempts, d.time, d.skills]);
  return ContentService.createTextOutput('ok');
}
function doGet() { return ContentService.createTextOutput('Nugget results endpoint is running.'); }

Each finished game adds a row: date, name, activity, game mode, pairs matched, attempts, time and skills. In two-player games each player gets a row.
